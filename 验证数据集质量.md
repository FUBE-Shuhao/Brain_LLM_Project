### 5.7



用tar指令打包了与FD值相关的文件



用generate_excel.py处理了这些文件，生成了xlsx表格,表格记录了每一个受试者的每一次测试的平均FD值，设定高于0.2的会被标亮，最终标亮sub-14,sub-31,但与老师讨论以后还是决定只剔除论文里写的sub-21,sub-52.





```python
import pandas as pd
import glob
import os
from collections import defaultdict

# ================= 配置区 =================
# 1. 你的数据路径
data_dir = r"C:\FMRI\QC" 
# 2. 输出文件名
output_excel = "fMRI_Detailed_Motion_Report.xlsx"
# ==========================================

# 查找所有 tsv 文件
search_pattern = os.path.join(data_dir, "**", "*desc-confounds_timeseries.tsv")
tsv_files = glob.glob(search_pattern, recursive=True)

# 核心存储结构：{受试者ID: {Run编号: Mean_FD}}
storage = defaultdict(dict)

print(f"正在分析 {len(tsv_files)} 个文件...")

for file in tsv_files:
    filename = os.path.basename(file)
    parts = filename.split('_')
    sub_id = parts[0]
    
    # 提取 Run 编号（如 run-1, run-2），如果没有则默认为 run-1
    run_info = next((p for p in parts if 'run-' in p), "run-1")
    
    try:
        df = pd.read_csv(file, sep='\t')
        fd_values = df['framewise_displacement'].dropna()
        if not fd_values.empty:
            storage[sub_id][run_info] = fd_values.mean()
    except:
        continue

# 将嵌套字典转换为 DataFrame
# 每一行是一个 Subject，每一列是一个 Run
rows = []
for sub_id, runs in storage.items():
    row_dict = {'Subject_ID': sub_id}
    row_dict.update(runs)
    rows.append(row_dict)

df_final = pd.DataFrame(rows)

# 排序：先按受试者名字排，再整理 Run 的列顺序
sub_col = ['Subject_ID']
run_cols = sorted([c for c in df_final.columns if c != 'Subject_ID'])
df_final = df_final[sub_col + run_cols]

# 计算 Avg_Mean_FD（排除 Subject_ID 列求横向平均）
df_final['Avg_Mean_FD'] = df_final[run_cols].mean(axis=1)

# 添加质量判定
def judge(val):
    if val > 0.3: return "Exclude (Severe)"
    if val > 0.2: return "Warning (High)"
    return "Pass"

df_final['Quality_Status'] = df_final['Avg_Mean_FD'].apply(judge)

# 排序
df_final = df_final.sort_values('Subject_ID')

# 写入 Excel 并美化
with pd.ExcelWriter(output_excel, engine='openpyxl') as writer:
    df_final.to_excel(writer, sheet_name='Motion_Report', index=False)
    
    # 自动调整列宽和颜色
    ws = writer.book['Motion_Report']
    from openpyxl.styles import PatternFill, Font, Alignment
    
    # 表头美化
    header_fill = PatternFill(start_color="4F81BD", end_color="4F81BD", fill_type="solid")
    for cell in ws[1]:
        cell.font = Font(bold=True, color="FFFFFF")
        cell.fill = header_fill
        cell.alignment = Alignment(horizontal='center')

    # 根据 FD 值标色
    red_fill = PatternFill(start_color="FFC7CE", end_color="FFC7CE", fill_type="solid")
    yellow_fill = PatternFill(start_color="FFEB9C", end_color="FFEB9C", fill_type="solid")
    
    avg_col_idx = df_final.columns.get_loc('Avg_Mean_FD') + 1
    for row in ws.iter_rows(min_row=2, max_row=ws.max_row):
        avg_val = row[avg_col_idx-1].value
        if avg_val > 0.3:
            for cell in row: cell.fill = red_fill
        elif avg_val > 0.2:
            for cell in row: cell.fill = yellow_fill

print(f"\n✅ 报表生成成功！共统计 {len(df_final)} 名受试者。")
print(f"输出路径: {os.path.abspath(output_excel)}")
```
