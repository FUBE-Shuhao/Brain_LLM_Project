import pandas as pd

excel_path = "/home/sh1/rawdata/text_data.xlsx"
output_path = "reading_brain_sentences.txt"

try:
    print(f"正在读取 {excel_path} ...")
    xl = pd.ExcelFile(excel_path)
    
    valid_sheets = [s for s in xl.sheet_names if s.strip().lower() != 'sheet']
    print(f"🎯 锁定 {len(valid_sheets)} 篇文章: {valid_sheets}\n")
    
    with open(output_path, 'w', encoding='utf-8') as f:
        for i, sheet_name in enumerate(valid_sheets):
            df = pd.read_excel(excel_path, sheet_name=sheet_name)
            
            # 智能定位：哪一列是单词？哪一列是编号(t.xx.xx)？
            id_col = None
            word_col = None
            
            for col in df.columns:
                sample_vals = df[col].dropna().astype(str).head(20).tolist()
                # 发现 t.01 这种格式的，就是句子编号列
                if any('t.0' in val or 's.0' in val for val in sample_vals):
                    id_col = col
                # 发现平均长度短于 15 且不是编号的，大概率是单词列
                elif any(val.isalpha() for val in sample_vals) and df[col].astype(str).str.len().mean() < 15:
                    if 'id' not in str(col).lower():
                        word_col = col
                        
            print(f"[{sheet_name}] 识别结果: 单词列=[{word_col}], 句子编号列=[{id_col}]")
            
            # 核心逻辑：组装句子
            current_id = None
            current_words = []
            count = 0
            
            for _, row in df.iterrows():
                sent_id = str(row[id_col]).strip()
                word = str(row[word_col]).strip()
                
                if word == 'nan' or not word:
                    continue
                    
                if current_id is None:
                    current_id = sent_id
                    
                # 如果遇到了新的句子编号，就把上一个句子拼起来写入
                if sent_id != current_id:
                    full_sentence = " ".join(current_words)
                    # 修复标点符号多余的空格 (比如 "word ." 修复为 "word.")
                    full_sentence = full_sentence.replace(" .", ".").replace(" ,", ",").replace(" ?", "?").replace(" !", "!")
                    f.write(full_sentence + "\n")
                    count += 1
                    
                    current_words = [word] # 开启新句子
                    current_id = sent_id
                else:
                    current_words.append(word)
                    
            # 把最后收集到的一句话写进去
            if current_words:
                full_sentence = " ".join(current_words)
                full_sentence = full_sentence.replace(" .", ".").replace(" ,", ",").replace(" ?", "?").replace(" !", "!")
                f.write(full_sentence + "\n")
                count += 1
                
            print(f"  -> 成功重构并写入 {count} 句话。\n")
            
            # 文章之间插入空行
            if i < len(valid_sheets) - 1:
                f.write("\n")
                
    print(f"✅ 完美提取！所有碎片单词已拼接成完整句子，文件已生成: {output_path}")

except Exception as e:
    print(f"❌ 运行崩溃: {e}")
